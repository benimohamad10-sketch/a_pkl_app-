Backend Google Apps Script (`Code.gs`)

Salin kode berikut ke Google Apps Script Spreadsheet Anda (Extensions > Apps Script) untuk membuat 8 Sheet otomatis dan API integrasi SIPKL: 

// ==========================================
// KODE BACKEND GOOGLE APPS SCRIPT SIPKL SMK TARUNA KARYA
// ==========================================

function setupDatabaseSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetNames = [
    "DATA SISWA",
    "PENDAFTARAN PKL",
    "MENCARI SENDIRI",
    "DICARIKAN SEKOLAH",
    "IKUT ORANG TUA/SAUDARA",
    "JADWAL PKL",
    "MONITORING PKL",
    "REKAPITULASI"
  ];
  
  sheetNames.forEach(function(name) {
    if (!ss.getSheetByName(name)) {
      ss.insertSheet(name);
    }
  });
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSiswa = ss.getSheetByName("DATA SISWA");
  var data = sheetSiswa.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetPendaftaran = ss.getSheetByName("PENDAFTARAN PKL");
  sheetPendaftaran.appendRow([
    new Date(),
    params.nis,
    params.nama,
    params.kelas,
    params.jurusan,
    params.kategori,
    params.perusahaan,
    params.statusTempat,
    params.statusPelaksanaan
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "SUCCESS"})).setMimeType(ContentService.MimeType.JSON);
}
Mencakup pembuatan sheet: DATA SISWA, PENDAFTARAN PKL, MENCARI SENDIRI, DICARIKAN SEKOLAH, IKUT ORANG TUA, JADWAL, MONITORING, REKAPITULASI
